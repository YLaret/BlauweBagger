CREATE TABLE DEVICE (
    DeviceID INTEGER PRIMARY KEY AUTOINCREMENT,
    SwitchID INTEGER,
    Name TEXT NOT NULL,
    Description TEXT,
    FOREIGN KEY (SwitchID) REFERENCES SWITCH(SwitchID)
);

CREATE TABLE SWITCH (
    SwitchID INTEGER PRIMARY KEY AUTOINCREMENT,
    Name TEXT NOT NULL,
    GPIO INTEGER NOT NULL
);

CREATE TABLE METER (
    MeterID INTEGER PRIMARY KEY AUTOINCREMENT,
    Name TEXT NOT NULL,
    Description TEXT,
    Value REAL NOT NULL
);

CREATE TABLE STAGE (
    StageID INTEGER PRIMARY KEY AUTOINCREMENT,
    Name TEXT NOT NULL,
    SwitchIDS TEXT,
    Time REAL NOT NULL
);

CREATE TABLE PROGRAM (
    ProgramID INTEGER PRIMARY KEY AUTOINCREMENT,
    Name TEXT NOT NULL,
    Description TEXT,
    StageIDS TEXT NOT NULL
);

CREATE TABLE MACHINESTATUS (
    ProgramID INTEGER NOT NULL,
    Pause INTEGER NOT NULL CHECK (Pause IN (0,1,2,14)),
    ProgramRunTime REAL NOT NULL,
    PRIMARY KEY (ProgramID),
    FOREIGN KEY (ProgramID) REFERENCES PROGRAM(ProgramID)
);

CREATE TABLE FORCE (
    SwitchIDS TEXT
);

CREATE TABLE METERRULES (
    MeterRuleID INTEGER PRIMARY KEY AUTOINCREMENT,
    MeterID INTEGER,
    MeterThreshold REAL NOT NULL,
    MeterThresholdGEQ INTEGER NOT NULL,
    SwitchID INTEGER,
    SwitchBool INTEGER NOT NULL,
    Stage INTEGER NOT NULL,
    FOREIGN KEY (MeterID) REFERENCES METER(MeterID)
    FOREIGN KEY (SwitchID) REFERENCES SWITCH(SwitchID)
);
